# OpenRotas Desktop — dependências da EDIÇÃO DESKTOP.
# Inclui tudo da aplicação web (reaproveitada por inteiro) + extras só do desktop.
# A raiz do repositório tem o requirements.txt da web; aqui referenciamos e acrescentamos.

-r ../requirements.txt

# --- Janela nativa com CHROMIUM EMBARCADO (preferida) ---
# PySide6/QtWebEngine traz o próprio motor de renderização DENTRO do executável: a janela
# nativa abre sem depender de nada instalado no Windows (sem WebView2, sem navegador, sem
# "localhost solto"). É o caminho primário (desktop/app/janela.py).
PySide6>=6.6            # inclui QtWebEngine (Chromium) via PySide6-Addons

# --- Janela nativa ALTERNATIVA (fallback leve, se o QtWebEngine não estiver disponível) ---
pywebview>=5.1          # usa o WebView2 (Edge) no Windows — leve, nativo
# O backend EdgeChromium (WebView2) do pywebview no Windows conversa com o runtime via
# pythonnet (módulo 'clr'). Só é usado no Windows; em Linux/macOS é ignorado pelo marcador.
pythonnet>=3.0.3 ; platform_system == "Windows"

# --- Diagnóstico de hardware (dimensionamento automático de workers/cache) ---
psutil>=5.9

# --- Notificação nativa do Windows (toast "estudo concluído") ---
# Pura-Python (sem extensões C); se ausente, notificacao.py cai no PowerShell/WinRT.
winotify>=1.1 ; platform_system == "Windows"

# --- Empacotamento (gera o executável onedir; o instalador é feito pelo Inno Setup) ---
pyinstaller>=6.6
